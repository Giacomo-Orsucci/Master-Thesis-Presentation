\begin{frame}{Neutral Atom Quantum Computing}
    \framesubtitle{Context}
    \begin{itemize}
        \item \textbf{The Paradigm:} Neutral atoms trapped in optical tweezers act as qubits, offering long coherence times and dynamic 2D/3D topologies.
        \item \textbf{Analog Mode \& QUBO:} The physical dynamics (Ising Hamiltonian) perfectly map to Quadratic Unconstrained Binary Optimization (QUBO) problems.
        \item \textbf{The Hardware Limit:} Exciting atoms to the Rydberg state induces the \textit{Rydberg blockade}. This naturally enforces constraints, making it ideal for Maximum Independent Set (MIS) problems on Unit Disk Graphs (UDG).
    \end{itemize}
\end{frame}

\begin{frame}{QUBO problems}
    \framesubtitle{QUBO formulation}
    
\begin{equation}
    \min_{x \in \{0,1\}^n} f(x) = x^T Q x = \sum_{i=1}^{n} \sum_{j=1}^{n} Q_{i,j} x_i x_j
\end{equation}

where:
\begin{itemize}
    \item $x = (x_1, x_2, \dots, x_n)$ is a vector of $n$ binary decision variables;
    \item $Q$ is an $n \times n$ square matrix of real constants detailing the problem's costs and interactions.
\end{itemize}

A key mathematical property of binary variables is that $x_i^2 = x_i$ for any $x_i \in \{0,1\}$. This elegantly allows any linear term of the objective function to be mapped directly onto the main diagonal of the $Q$ matrix ($Q_{i,i}$), while the off-diagonal elements ($Q_{i,j}$ for $i \neq j$) encode the quadratic interactions, or "penalties," between pairs of variables.
\end{frame}


\begin{frame}{Physical Formulation}
\framesubtitle{QUBO to Ising Hamiltonian Mapping}
    The energy of the system is described the Ising Hailtonian and The QUBO framework is an isomorphism of it: 
    
    \begin{equation}
    \hat{\mathcal{H}}(t)=\hbar\sum_{i=1}^{N}\left(\frac{\Omega(t)}{2}\hat{\sigma}_{i}^{x}-\delta(t)\hat{n}_{i}\right)+\sum_{i<j}\frac{C_{6}}{|r_{i}-r_{j}|^{6}}\hat{n}_{i}\hat{n}_{j}
\end{equation}

\begin{itemize}
    \item $\hbar$ is the reduced Planck constant, which scales the angular frequencies of the driving laser fields ($\Omega$ and $\delta$) into actual energy terms;    \item $N$ is the total number of atoms (qubits) in the register;
    \item $\Omega(t)$ is the time-dependent Rabi frequency, representing the amplitude of the laser exciting the atoms;
    \item $\hat{\sigma}_{i}^{x}$ is the Pauli-X operator acting on atom $i$, which drives the transition between the ground state $|0\rangle$ and the Rydberg state $|1\rangle$;
    \item $\delta(t)$ is the detuning parameter of the laser frequency;
   
\end{itemize}

\end{frame}

\begin{frame}{Physical Formulation}

\framesubtitle{QUBO to Ising Hamiltonian Mapping}

  \begin{equation}
    \hat{\mathcal{H}}(t)=\hbar\sum_{i=1}^{N}\left(\frac{\Omega(t)}{2}\hat{\sigma}_{i}^{x}-\delta(t)\hat{n}_{i}\right)+\sum_{i<j}\frac{C_{6}}{|r_{i}-r_{j}|^{6}}\hat{n}_{i}\hat{n}_{j}
    \end{equation}

    \vspace{0.3cm}

    \begin{itemize}
         \item $\hat{n}_{i}$ is the occupation number operator (defined as $|1\rangle_i\langle 1|_i$) which evaluates to 1 if atom $i$ is in the Rydberg state, and 0 otherwise;
        \item $r_i$ is the spatial position of atom $i$;
        \item $C_6$ is the Van der Waals interaction coefficient, which is a constant dependent on the specific atomic species and platform used.

    \end{itemize}
\end{frame}

\begin{frame}{The Embedding Problem}
    \framesubtitle{Hardware Constraints (Pasqal \textit{Jade})}
    To execute a QUBO problem, logical variables must be mapped onto physical 2D coordinates. This spatial embedding is an $\mathsf{NP}$-Hard problem itself.
    \vspace{0.5cm}
    
    \textbf{Hardware Constraints (Pasqal \textit{Jade}):}
    \begin{itemize}
        \item \textbf{Maximum Radius ($R_{\text{max}}$):} Register confined within $50.0 \, \mu\text{m}$.
        \item \textbf{Minimum Distance ($d_{\text{min}}$):} Atoms cannot be closer than $4.0 \, \mu\text{m}$ (diffraction limit).
        \item \textbf{Global Laser Parameterization:} Uniform detuning and Rabi frequency for all atoms.
    \end{itemize}
\end{frame}

\begin{frame}{The Scalability Wall}
\framesubtitle{The Operational Breaking Point}
    \textbf{The Operational Breaking Point} \\
    Despite hardware resilience in quantum emulation, classical monolithic embedding heuristics face the \textit{spatial curse of dimensionality}.
    
    \vspace{0.3cm}
    \begin{itemize}
        \item GA pipeline fails on $15 \times 15$ instances.
        \item SA pipeline hits a computational wall on dense $20 \times 20$ matrices, leading to degraded spatial fitness and noisy quantum readouts.
        \item \textbf{Conclusion:} With better classical hardware we can surely scale more, but the embedding of larger problems benefits from a paradigm shift.
    \end{itemize}
\end{frame}

\begin{frame}{Conclusions}
\framesubtitle{Take-Home Messages}
    \begin{itemize}
        \item \textbf{Current Hardware Viability:} Neutral Atom platforms have proven highly promising, capable of tackling combinatorial optimization problems beyond toy-dimension instances.
        \item \textbf{The Necessity of Partitioning:} Monolithic embeddings hit strict physical ceilings. The \textit{Divide et Impera} architecture bypasses continuous spatial constraints by transforming intractable global configurations into manageable micro-tasks.
        \item \textbf{Future-Proof Modularity:} The distributed approach is inherently modular. As hardware capabilities expand, the architecture scales seamlessly without requiring structural redesigns.
    \end{itemize}
\end{frame}

% SLIDE: Future Outlook 1
\begin{frame}{Future Outlook}
\framesubtitle{Moving Beyond Analog Emulation}
    \begin{itemize}
        \item \textbf{Direct Physical Deployment:} Overcoming the current reliance on HPC emulators by submitting JSON job registries directly to a physical Neutral Atom QPU, evaluating empirical resilience against raw physical noise.
        \item \textbf{Advanced Laser Pulse Optimization:} Exploring Optimal Control Theory (OCT) or Analog QAOA to dynamically shape the Rabi frequency and detuning profiles, minimizing diabatic transitions and boosting ground-state probability.
        \item \textbf{Alternative Problem Formulations:} Expanding the orchestrator to automatically translate higher-level combinatorial problems (e.g., Integer Programming) into platform-compliant UDG topologies.
    \end{itemize}
\end{frame}

        \item \textbf{Quantum-Native Boundary Resolution:} "Freezing" internal atoms that reached an optimal state and isolating strictly the conflicting \textit{border graph}. Resubmitting this border to the QPU fully delegates conflict resolution to the quantum hardware.

\begin{frame}{Appendix: Simulated Annealing Engine}
\framesubtitle{Energy Formulation and Spatial Perturbations}
    SA operates as a probabilistic, single-trajectory metaheuristic. It optimizes the continuous spatial layout by iteratively minimizing an internal energy $E(S)$:
    $$ E(S) = E_{active} + E_{crosstalk} + P_{geom} $$

    \vspace{0.2cm}
    \textbf{1. Geometric Neighborhood Moves ($S \to S^{\prime}$):}
    \begin{itemize}
        \item \textbf{Jitter ($60\%$):} Micro-displacement of a single atom. The maximum radius scales proportionally with the current temperature.
        \item \textbf{Swap ($20\%$):} The coordinates of two distinct atoms are perfectly swapped, resolving topological entanglements.
        \item \textbf{Jump ($20\%$):} An atom is randomly replaced anywhere within the permissible $R_{max}$ boundary for global exploration.
    \end{itemize}

    \vspace{0.2cm}
    \textbf{2. Thermodynamic Evolution:} \\
    Transitions to a worse configuration ($\Delta E > 0$) are accepted following the Boltzmann distribution to escape local minima:
    $$ P(S \to S^{\prime}) = e^{-\frac{\Delta E}{T}} $$
\end{frame}

\begin{frame}{Appendix: Fitness Function Formulation}
\framesubtitle{Improved Topological MAE (1/2)}
    The custom fitness function evaluates the geometric quality of a candidate layout and its structural isomorphism to the target QUBO matrix.
    \vspace{0.2cm}

    \textbf{1. Physical Potential Clipping:}\\
    As $r_{ij} \to 0$, the physical potential explodes. To prevent numerical instability, the effective potential $\tilde{V}_{ij}$ is bounded using a saturation threshold $n=5$:
    $$ \tilde{V}_{ij} = \min\left(\frac{C_6}{r_{ij}^6}, n \cdot Q_{max}\right) $$
    
    \vspace{0.2cm}

    \textbf{2. Topological Base Error ($E_{base}$):}\\
    Edges are partitioned into active ($> \epsilon$) and inactive ($\leq \epsilon$) sets using a tolerance $\epsilon=10^{-5}$. The error penalizes deviations on active bonds and unwanted crosstalk on inactive ones:
    $$ E_{base} = \sum_{(i,j) \in E_{active}} |\tilde{V}_{ij} - Q_{ij}| + \lambda \sum_{(i,j) \in E_{inactive}} \tilde{V}_{ij} $$
    The amplification multiplier $\lambda = 2.0$ forces the algorithm to move unconnected nodes apart.
\end{frame}

% SLIDE: Appendix - Fitness Function (2/2)
\begin{frame}{Appendix: Fitness Function Formulation}
\framesubtitle{Improved Topological MAE (2/2)}
    \textbf{3. Geometric Hardware Penalty ($P$):}\\
    The layout must strictly respect the optical tweezers' diffraction limits ($d_{min} = 4.0 \, \mu\text{m}$) and the global laser's maximum radius ($R_{max} = 50.0 \, \mu\text{m}$):
    $$ P = K \left( \sum_{i<j} \max(0, d_{min} - r_{ij}) + \sum_{i} \max(0, ||r_i|| - R_{max}) \right) $$
    A massive penalty factor $K = 10^5$ immediately drops physically unfeasible candidates from the evolutionary pool.
    
    \vspace{0.4cm}

    \textbf{4. Final Fitness Value ($F$):}\\
    Because Genetic Algorithms natively maximize a score, the accumulated errors and penalties are inverted:
    $$ F = \frac{1}{E_{base} + P + \delta} $$
    A minimal constant $\delta = 10^{-6}$ is introduced to prevent zero-division crashes.
\end{frame}

% SLIDE: Appendix - Spectral Clustering (1/2)
\begin{frame}{Appendix: Spectral Clustering}
\framesubtitle{Graph Translation and Laplacian Matrix}
    Randomly splitting the QUBO matrix would sever crucial quadratic interactions. Instead, we translate it into an Affinity Matrix $A$:
    $$ A_{ij} = |Q_{ij}|(1 - \delta_{ij}) $$
    The Kronecker delta ($\delta_{ij}$) forces $A_{ii} = 0$, ignoring linear biases to focus exclusively on the pure topological interconnectivity.
    
    \vspace{0.4cm}
    Next, we compute the Degree Matrix $D$ and the unnormalized Graph Laplacian $L$:
    $$ D_{ii} = \sum_{j} A_{ij} $$
    $$ L = D - A $$
    \vspace{0.2cm}
    
    The eigenvector associated with the second smallest eigenvalue of $L$ (the \textit{Fiedler vector}) provides the optimal continuous relaxation for the discrete graph partitioning problem.
\end{frame}

% SLIDE: Appendix - Spectral Clustering (2/2)
\begin{frame}{Appendix: Spectral Clustering}
\framesubtitle{Algorithm and Recursive Size-Capping}
    
    To enforce the hardware's maximum qubit capacity, we dynamically calculate the number of target clusters $k$:
    $$ k = \max(2, \text{round}(N / \text{target\_capacity})) $$
    
    \vspace{0.2cm}
    \begin{block}{Algorithm: Spectral Graph Clustering}
        \begin{enumerate}
            \item Create a sparsified similarity graph $G$ (represented by matrix $A$).
            \item Compute the graph Laplacian for $G$, denoted as $L$.
            \item Create a matrix $V$ consisting of the first $k$ eigenvectors of $L$.
            \item Apply K-means clustering on the rows of $V$ to obtain the $k$ discrete clusters.
        \end{enumerate}
    \end{block}
    
    \vspace{0.2cm}
    \textbf{Recursive execution:} If any resulting partition still exceeds the maximum hardware capacity, the algorithm recursively calls itself exclusively on that overgrown sub-graph.
\end{frame}

\begin{frame}{Appendix: Global Merging Refinement}
\framesubtitle{Low-Temperature SA Post-Processing}
    While the deterministic greedy merging efficiently resolves boundary overlaps, it is prone to settling into sub-optimal configurations at the cluster intersections. 
    \vspace{0.3cm}
    
    To refine the global state, the pipeline applies a final localized perturbation using Simulated Annealing:
    
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Hot-Start Initialization:} Unlike standard SA, this phase does not begin from a random state. It directly injects the greedy-merged bitstring as the starting configuration.
        \item \textbf{Restricted Thermal Range:} By bounding the inverse temperature parameter ($\beta$) to a narrow, high-value range, the engine performs strictly localized, low-temperature micro-adjustments.
        \item \textbf{Structure Preservation:} This thermodynamic fine-tuning effectively squeezes out the remaining topological errors without destroying the highly optimized global structure assembled in the preceding steps.
    \end{itemize}
\end{frame}
